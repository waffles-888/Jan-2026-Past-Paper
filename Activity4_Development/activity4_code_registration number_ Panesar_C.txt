import randomimport stringimport timemin_characters = 5max_characters = 15while True:    first_name = input("Please enter your first name: ").strip()    if first_name == "":        print("First name cannot be left blank")        continue    breakwhile True:    last_name = input("Please enter your last name: ").strip()    if last_name == "":        print("Last name cannot be left blank")        continue    breakwhile True:    num_char = input("How many characters would you like (5 - 15): ").strip()    if not num_char.isdigit():        print("Please enter a number between 5 Ð 15")        continue    num_char = int(num_char)    if num_char < min_characters or num_char > max_characters:        print("Please enter a number between 5 Ð 15")        continue    breakcharacters = ""for i in range(num_char):    characters = characters + random.choice(string.ascii_letters + string.digits)print("Enter the characters displayed as fast as possible")print(characters)start_time = time.time()char_answer = input("> ")stop_time = time.time()total_time = round(stop_time - start_time, 2)correct = 0for i in range(num_char):    if i < len(char_answer) and char_answer[i] == characters[i]:        correct = correct + 1print(first_name + " " + last_name + ": You completed this in " + str(total_time) + " seconds.")print("You had " + str(correct) + " out of " + str(num_char) + " correct.")
